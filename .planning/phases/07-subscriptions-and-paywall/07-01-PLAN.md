   ```typescript
   // Phase 7: RevenueCat webhook handler (D-19)
   // Receives subscription events and updates profiles.subscription_status
   // Source: https://www.revenuecat.com/docs/integrations/webhooks
   //
   // SECURITY: Authorization header verified against RC_WEBHOOK_SECRET (T-07-02)
   // IDEMPOTENCY: Uses state-driven update (not event-type inference) to handle
   //   duplicate webhook deliveries safely (RESEARCH Pitfall 7)
   //
   // RevenueCat webhook payload fields used:
   //   event.type — INITIAL_PURCHASE, RENEWAL, CANCELLATION, EXPIRATION, etc.
   //   event.app_user_id — Supabase UUID (set via Purchases.shared.logIn())
   //   event.id — unique event ID for idempotency tracking

   import { serve } from "https://deno.land/std@0.177.0/http/server.ts"
   import { createClient } from "https://esm.sh/@supabase/supabase-js@2"

   const ACTIVE_EVENTS = new Set([
     "INITIAL_PURCHASE",
     "RENEWAL",
     "UNCANCELLATION",
     "BILLING_ISSUE_RESOLVED",
     "PRODUCT_CHANGE",
     "TRANSFER",
   ])

   const GRACE_EVENTS = new Set([
     "BILLING_ISSUE",
   ])

   const INACTIVE_EVENTS = new Set([
     "CANCELLATION",
     "EXPIRATION",
     "SUBSCRIPTION_PAUSED",
   ])

   serve(async (req: Request) => {
     // 1. Verify authorization header (T-07-02)
     const authHeader = req.headers.get("Authorization")
     const secret = Deno.env.get("RC_WEBHOOK_SECRET")
     if (!secret || authHeader !== `Bearer ${secret}`) {
       return new Response("Unauthorized", { status: 401 })
     }

     // 2. Parse event
     const event = await req.json()
     const appUserId: string = event.app_user_id
     const eventType: string = event.type

     // 3. Reject anonymous IDs -- they indicate logIn() was not called (Pitfall 1)
     if (!appUserId || appUserId.startsWith("$RCAnonymousID")) {
       console.error(`Webhook received anonymous ID: ${appUserId}. RevenueCat logIn() not called with Supabase UUID.`)
       // Return 200 to prevent RevenueCat retries on a permanently bad payload
       return new Response("Anonymous ID rejected", { status: 200 })
     }

     // 4. Determine new subscription status (state-driven, not event-only -- Pitfall 7)
     let status: string
     if (ACTIVE_EVENTS.has(eventType)) {
       status = "subscribed"
     } else if (GRACE_EVENTS.has(eventType)) {
       status = "grace_period"
     } else if (INACTIVE_EVENTS.has(eventType)) {
       status = "free"
     } else {
       // Unknown event type -- log and acknowledge
       console.log(`Unknown event type: ${eventType}`)
       return new Response("OK", { status: 200 })
     }

     // 5. Update profiles table
     const supabase = createClient(
       Deno.env.get("SUPABASE_URL")!,
       Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!
     )

     const { error } = await supabase
       .from("profiles")
       .update({ subscription_status: status })
       .eq("id", appUserId)

     if (error) {
       console.error(`Failed to update profile ${appUserId}: ${error.message}`)
       // Return 500 so RevenueCat retries (up to 5x)
       return new Response("Update failed", { status: 500 })
     }

     // 6. Must return 200 within 60s or RevenueCat retries
     return new Response("OK", { status: 200 })
   })
   ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| iOS Client -> RevenueCat SDK | Client sends purchase requests; SDK validates with Apple and grants entitlements |
| RevenueCat -> Supabase Edge Function | Webhook delivers subscription events; must verify auth header |
| iOS Client -> Supabase | Client reads profiles; subscription_status is server-authoritative, not client-set |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-07-01 | Spoofing | AppState.isSubscribed | mitigate | Client isSubscribed is UX gate only; backend AI proxy Edge Functions must gate on profiles.subscription_status (D-19), never trust client |
| T-07-02 | Spoofing | revenuecat-webhook Edge Function | mitigate | Verify Authorization header against RC_WEBHOOK_SECRET before processing any event; reject with 401 if missing/wrong |
| T-07-03 | Spoofing | Webhook app_user_id | mitigate | Reject payloads where app_user_id starts with $RCAnonymousID — indicates logIn() was not called (Pitfall 1) |
