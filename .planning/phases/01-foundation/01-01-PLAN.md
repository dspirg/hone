       ```sql
       -- Phase 1: profiles table, RLS, auto-create trigger
       -- Requirements: AUTH-01, AUTH-02 (profile created on sign-up)
       -- Decisions: D-08 (auth + profiles only), D-09 (columns), D-10 (RLS)

       -- 1. Create profiles table (D-09)
       CREATE TABLE public.profiles (
           id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,
           display_name TEXT,
           avatar_url TEXT,
           onboarding_completed BOOLEAN NOT NULL DEFAULT FALSE,
           subscription_status TEXT NOT NULL DEFAULT 'free'
               CHECK (subscription_status IN ('free', 'subscribed')),
           created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
           PRIMARY KEY (id)
       );

       -- 2. Enable RLS (D-10)
       ALTER TABLE public.profiles ENABLE ROW LEVEL SECURITY;

       -- 3. RLS policies (D-10) -- users can only read/write their own row
       CREATE POLICY "Users can view own profile"
           ON public.profiles FOR SELECT
           USING (auth.uid() = id);

       CREATE POLICY "Users can update own profile"
           ON public.profiles FOR UPDATE
           USING (auth.uid() = id);

       -- 4. Trigger function -- SECURITY DEFINER required for cross-schema INSERT (Pitfall 5)
       CREATE OR REPLACE FUNCTION public.handle_new_user()
       RETURNS TRIGGER
       LANGUAGE plpgsql
       SECURITY DEFINER SET search_path = ''
       AS $$
       BEGIN
           INSERT INTO public.profiles (id, display_name, avatar_url)
           VALUES (
               NEW.id,
               NEW.raw_user_meta_data ->> 'display_name',
               NEW.raw_user_meta_data ->> 'avatar_url'
           );
           RETURN NEW;
       EXCEPTION WHEN OTHERS THEN
           RAISE WARNING 'handle_new_user: %', SQLERRM;
           RETURN NEW;
       END;
       $$;

       -- 5. Attach trigger
       CREATE TRIGGER on_auth_user_created
           AFTER INSERT ON auth.users
           FOR EACH ROW EXECUTE PROCEDURE public.handle_new_user();
       ```

       ```swift
       import Observation
       import Supabase

       @Observable
       final class AppState {
           var isAuthenticated: Bool = false
           var currentUser: User? = nil
           var showPasswordResetForm: Bool = false

           func listenForAuthChanges() async {
               for await (event, session) in supabase.auth.authStateChanges {
                   switch event {
                   case .initialSession, .signedIn, .tokenRefreshed, .userUpdated:
                       self.isAuthenticated = session != nil
                       self.currentUser = session?.user
                   case .signedOut:
                       self.isAuthenticated = false
                       self.currentUser = nil
                   case .passwordRecovery:
                       self.showPasswordResetForm = true
                   default:
                       break
                   }
               }
           }
       }
       ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| iOS client -> Supabase Auth | Auth credentials (email/password, Apple ID token) cross network boundary |
| Supabase Auth -> PostgreSQL | Trigger crosses schema boundary (auth -> public) |
| xcconfig -> binary | Env vars embedded in Info.plist at build time |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-01 | Information Disclosure | SupabaseClient.swift | mitigate | Only anon key in binary via xcconfig; never service_role key; prod key in Prod.xcconfig (not committed) |
| T-01-02 | Information Disclosure | JWT storage | mitigate | `KeychainLocalStorage(service: bundleIdentifier)` with `.whenUnlockedThisDeviceOnly` -- never UserDefaults |
| T-01-03 | Tampering | PKCE flow | mitigate | `flowType: .pkce` set explicitly in SupabaseClientOptions; never override to `.implicit` |
| T-01-04 | Elevation of Privilege | profiles table | mitigate | RLS enabled with `auth.uid() = id` policies for SELECT and UPDATE; no INSERT policy (trigger bypasses RLS as postgres) |
| T-01-05 | Denial of Service | Trigger failure | mitigate | `EXCEPTION WHEN OTHERS` block in trigger prevents sign-up rollback on profile creation failure |
| T-01-06 | Tampering | Deep link URL scheme | accept | Custom URL scheme `workout://` is not cryptographically verified; acceptable because PKCE code verifier prevents replay -- the URL scheme merely delivers the callback |
</threat_model>
