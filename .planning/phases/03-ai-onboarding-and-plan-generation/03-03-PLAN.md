**PlanSSEClient.swift** — SSE streaming wrapper:

CRITICAL: Do NOT use `supabase.functions.invokeWithStreamedResponse`. This method does NOT forward the JWT auth token (Supabase Swift SDK issue #634). Build the URLRequest manually.

```swift
import Foundation
import Supabase

enum PlanSSEEvent {
    case token(String)       // partial JSON token chunk
    case completed(String)   // full accumulated JSON string
    case error(Error)
}

enum PlanSSEError: Error, LocalizedError {
    case notAuthenticated
    case invalidURL
    case streamFailed(statusCode: Int)
    case networkError(underlying: Error)

    var errorDescription: String? {
        switch self {
        case .notAuthenticated: return "Not authenticated"
        case .invalidURL: return "Invalid Edge Function URL"
        case .streamFailed(let code): return "Stream failed with status \(code)"
        case .networkError(let err): return err.localizedDescription
        }
    }
}

final class PlanSSEClient {
    private let supabaseURL: String

    init() {
        self.supabaseURL = Bundle.main.infoDictionary?["SUPABASE_URL"] as? String ?? ""
    }

    /// Streams plan generation tokens from the Edge Function.
    /// Returns an AsyncThrowingStream that yields partial JSON token strings.
    /// The final yield is the complete accumulated JSON string after [DONE].
    func streamPlan(profile: UserProfile) -> AsyncThrowingStream<PlanSSEEvent, Error> {
        AsyncThrowingStream { continuation in
            Task {
                do {
                    // 1. Get session token — MUST use supabase.auth.session, not SDK invoke
                    guard let session = try? await supabase.auth.session else {
                        continuation.finish(throwing: PlanSSEError.notAuthenticated)
                        return
                    }

                    // 2. Build manual URLRequest
                    guard let url = URL(string: "\(supabaseURL)/functions/v1/generate-plan") else {
                        continuation.finish(throwing: PlanSSEError.invalidURL)
                        return
                    }

                    var request = URLRequest(url: url)
                    request.httpMethod = "POST"
                    // CRITICAL: Manual auth header — SDK streaming path drops this
                    request.setValue("Bearer \(session.accessToken)", forHTTPHeaderField: "Authorization")
                    request.setValue("application/json", forHTTPHeaderField: "Content-Type")
                    request.httpBody = try JSONEncoder().encode(profile)

                    // 3. Stream bytes — URLSession.shared.bytes is iOS 15+ async
                    let (asyncBytes, response) = try await URLSession.shared.bytes(for: request)

                    if let httpResponse = response as? HTTPURLResponse,
                       httpResponse.statusCode != 200 {
                        continuation.finish(throwing: PlanSSEError.streamFailed(statusCode: httpResponse.statusCode))
                        return
                    }

                    // 4. Parse SSE lines — accumulate JSON tokens
                    var accumulatedJSON = ""

                    for try await line in asyncBytes.lines {
                        // SSE format: "data: {json_chunk}" or "data: [DONE]"
                        guard line.hasPrefix("data: ") else { continue }
                        let data = String(line.dropFirst(6))

                        if data == "[DONE]" {
                            // Stream complete — yield the full accumulated JSON
                            continuation.yield(.completed(accumulatedJSON))
                            break
                        }

                        // OpenAI streaming: each chunk is a ChatCompletion delta
                        // Extract the content delta from the chunk
                        if let chunkData = data.data(using: .utf8),
                           let chunk = try? JSONSerialization.jsonObject(with: chunkData) as? [String: Any],
                           let choices = chunk["choices"] as? [[String: Any]],
                           let delta = choices.first?["delta"] as? [String: Any],
                           let content = delta["content"] as? String {
                            accumulatedJSON += content
                            continuation.yield(.token(content))
                        }
                    }

                    continuation.finish()
                } catch {
                    continuation.finish(throwing: PlanSSEError.networkError(underlying: error))
                }
            }
        }
    }
}
```

IMPORTANT — Pitfall 3: Do NOT attempt to parse `accumulatedJSON` as WorkoutPlan during streaming. The partial JSON is not valid until `[DONE]`. Only parse the complete string yielded by `.completed(accumulatedJSON)`.

**PlanGenerationService.swift** — Orchestrates the full plan generation lifecycle:

```swift
import Foundation
import Supabase

enum GenerationState: Equatable {
    case idle
    case streaming(partialText: String)   // tokens arriving
    case completed(WorkoutPlan)           // plan decoded
    case error(String)                    // user-facing error message
}

@Observable
@MainActor
final class PlanGenerationService {
    var state: GenerationState = .idle
    var regenerationsRemaining: Int = 3   // D-09: 3 free regenerations

    private let sseClient = PlanSSEClient()
    private let repository = WorkoutPlanRepository()
    private var currentStreamTask: Task<Void, Never>?

    /// Generates a plan from the user profile.
    /// Handles silent auto-retry on first failure (D-16).
    /// On success: saves to Supabase + CoreData, updates onboarding flag.
    /// Strict ordering per Pitfall 4: (1) Supabase plan write, (2) CoreData write, (3) onboarding_completed flag.
    func generatePlan(profile: UserProfile, isRetry: Bool = false) {
        currentStreamTask?.cancel()
        state = .streaming(partialText: "")

        currentStreamTask = Task {
            do {
                var accumulatedText = ""

                for try await event in sseClient.streamPlan(profile: profile) {
                    if Task.isCancelled { return }

                    switch event {
                    case .token(let chunk):
                        accumulatedText += chunk
                        state = .streaming(partialText: accumulatedText)

                    case .completed(let fullJSON):
                        // Parse completed JSON into WorkoutPlan
                        guard let jsonData = fullJSON.data(using: .utf8) else {
                            throw PlanSSEError.networkError(underlying: NSError(domain: "PlanGen", code: -1))
                        }
                        let plan = try JSONDecoder().decode(WorkoutPlan.self, from: jsonData)

                        // Persist — STRICT ORDER per Pitfall 4:
                        // 1. Write plan to Supabase workout_plans
                        let supabaseId = try await savePlanToSupabase(plan: plan, profile: profile)

                        // 2. Write plan to CoreData
                        guard let userId = try? await supabase.auth.session.user.id.uuidString else { return }
                        try repository.deactivateAllPlans(userId: userId)
                        try repository.save(plan: plan, supabaseId: supabaseId, userId: userId)

                        // 3. Set onboarding_completed = true in profiles
                        try await setOnboardingCompleted()

                        state = .completed(plan)

                    case .error(let error):
                        throw error
                    }
                }
            } catch {
                if Task.isCancelled { return }

                // D-16: Silent auto-retry once on first failure
                if !isRetry {
                    generatePlan(profile: profile, isRetry: true)
                } else {
                    // Second failure — show error
                    state = .error("Something went wrong")
                }
            }
        }
    }

    /// Regenerates the plan. Decrements counter. Disabled at 0 per D-09.
    func regeneratePlan(profile: UserProfile) {
        guard regenerationsRemaining > 0 else { return }
        regenerationsRemaining -= 1
        generatePlan(profile: profile)
    }

    /// Whether regeneration is allowed
    var canRegenerate: Bool {
        regenerationsRemaining > 0
    }

    // MARK: - Private

    private func savePlanToSupabase(plan: WorkoutPlan, profile: UserProfile) async throws -> String {
        let userId = try await supabase.auth.session.user.id

        // Encode plan to JSONB
        let planJSON = try JSONEncoder().encode(plan)

        struct PlanInsert: Encodable {
            let user_id: String
            let plan_name: String
            let goal_summary: String
            let plan_json: String  // JSONB as string
            let days_per_week: Int
            let is_active: Bool
        }

        let insert = PlanInsert(
            user_id: userId.uuidString,
            plan_name: plan.planName,
            goal_summary: plan.goalSummary,
            plan_json: String(data: planJSON, encoding: .utf8) ?? "{}",
            days_per_week: profile.daysPerWeek,
            is_active: true
        )

        // Deactivate existing plans first
        try await supabase
            .from("workout_plans")
            .update(["is_active": false])
            .eq("user_id", value: userId.uuidString)
            .execute()

        // Insert new plan
        let response = try await supabase
            .from("workout_plans")
            .insert(insert)
            .select("id")
            .single()
            .execute()

        // Extract the new plan UUID
        if let result = try? JSONSerialization.jsonObject(with: response.data) as? [String: Any],
           let id = result["id"] as? String {
            return id
        }
        return UUID().uuidString  // fallback
    }

    private func setOnboardingCompleted() async throws {
        let userId = try await supabase.auth.session.user.id
        try await supabase
            .from("profiles")
            .update(["onboarding_completed": true])
            .eq("id", value: userId.uuidString)
            .execute()
    }

    /// Saves user profile to Supabase profiles table (called before plan generation).
    func saveProfile(_ profile: UserProfile) async throws {
        let userId = try await supabase.auth.session.user.id
        try await supabase
            .from("profiles")
            .update([
                "goal": profile.goal,
                "fitness_level": profile.fitnessLevel,
                "days_per_week": profile.daysPerWeek,
                "equipment": profile.equipment,
                "injuries": profile.injuries
            ] as [String: AnyJSON])
            .eq("id", value: userId.uuidString)
            .execute()
    }
}
```
