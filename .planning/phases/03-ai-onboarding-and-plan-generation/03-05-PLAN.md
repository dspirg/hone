Add private method:
```swift
private func fetchOnboardingStatus() async {
    guard let userId = currentUser?.id else { return }
    do {
        struct ProfileRow: Decodable {
            let onboarding_completed: Bool
        }
        let response: ProfileRow = try await supabase
            .from("profiles")
            .select("onboarding_completed")
            .eq("id", value: userId.uuidString)
            .single()
            .execute()
            .value
        self.onboardingCompleted = response.onboarding_completed
    } catch {
        // Default to false — user will see onboarding
        self.onboardingCompleted = false
    }
}
```

Also add method to set the flag programmatically (called after plan generation):
```swift
func markOnboardingComplete() {
    self.onboardingCompleted = true
}
```

On sign-out, reset the flag:
```swift
case .signedOut:
    self.isAuthenticated = false
    self.currentUser = nil
    self.onboardingCompleted = false
```

**OnboardingFlowView** — New container view (add in WorkoutApp.swift below ContentView):
This view owns the flow from onboarding cards through plan generation to plan preview:
```swift
struct OnboardingFlowView: View {
    @Environment(AppState.self) var appState
    @State private var planService = PlanGenerationService()
    @State private var showPlanPreview = false
    @State private var userProfile: UserProfile?

    var body: some View {
        if showPlanPreview, let profile = userProfile {
            let viewModel = PlanPreviewViewModel(service: planService, profile: profile)
            PlanPreviewView(viewModel: viewModel, onStartTraining: {
                appState.markOnboardingComplete()
            })
        } else {
            OnboardingView(onComplete: { profile in
                userProfile = profile
                showPlanPreview = true
                // Start plan generation immediately per D-11
                planService.generatePlan(profile: profile)
            })
        }
    }
}
```

The flow: OnboardingView cards -> user completes -> `onComplete` fires with UserProfile -> transition to PlanPreviewView -> PlanGenerationService streams plan -> plan displays -> user taps "Start Training" -> `appState.markOnboardingComplete()` -> ContentView routes to MainTabView.

                            .font(.body)
                            .foregroundStyle(.secondary)
                            .multilineTextAlignment(.center)
                            .padding(.horizontal, 32)
                    }
                    .frame(maxWidth: .infinity)
                    .padding(.top, 48)
                }
            }
            .navigationTitle("Home")
            .task {
                await loadActivePlan()
            }
        }

    private func loadActivePlan() async {
        guard let userId = appState.currentUser?.id.uuidString else {
            isLoading = false
            return
        }
        do {
            let repo = WorkoutPlanRepository()
            activePlan = try repo.fetchActivePlan(userId: userId)
        } catch {
            // Fallback: no plan found; show empty state
        }
        isLoading = false
    }
}

        state.onboardingCompleted = true

    }

        state.onboardingCompleted = true
