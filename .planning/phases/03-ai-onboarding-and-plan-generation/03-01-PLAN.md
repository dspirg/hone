**Edge Function -- `supabase/functions/generate-plan/index.ts`:**

Create directory `supabase/functions/generate-plan/` and add `index.ts`. The function:

1. Reads JSON body: `{ goal, fitness_level, days_per_week, equipment, injuries }`.
2. Validates the request has an `Authorization` header. The iOS client sends `Bearer <token>` manually because the Supabase Swift SDK `invokeWithStreamedResponse` does NOT forward JWT (bug #634). Log a warning if no auth header is present.
3. Constructs a system prompt injecting user profile fields per D-12. The prompt must:
   - State the AI is a professional fitness coach generating a personalized plan
   - Include user goal, fitness level, days per week, equipment list, and injuries (if non-empty string)
   - Instruct the model to generate exactly `days_per_week` training days
   - Instruct each exercise to include a `rationale` explaining why it was chosen for this user's specific goal and available equipment per AIPL-02
   - Include safety guardrails: refuse medical diagnosis/treatment advice, recommend physician consult for pain/injury concerns per SAFE-02
   - If `injuries` is a non-empty string, include a section: "Areas to avoid or modify around: {injuries}"
   - If `injuries` is empty string, omit that section entirely
4. Calls `https://api.openai.com/v1/chat/completions` with:
   - `model: "gpt-4o-2024-08-06"` (minimum version for Structured Outputs)
   - `stream: true`
   - `response_format: { type: "json_schema", json_schema: { name: "workout_plan", strict: true, schema: planSchema } }`
5. The `planSchema` must have `additionalProperties: false` at EVERY object level (root, day, exercise) and ALL fields in `required` arrays. This is mandatory for OpenAI Structured Outputs strict mode -- omitting any level causes a 400 error. The `injuries` field is NOT in the output schema -- it only affects the system prompt. The schema:

import Foundation

struct WorkoutPlan: Codable, Equatable, Sendable {
    let planName: String
    let goalSummary: String
    let weeklyDays: [WorkoutDay]

    enum CodingKeys: String, CodingKey {
        case planName = "plan_name"
        case goalSummary = "goal_summary"

struct WorkoutDay: Codable, Equatable, Identifiable, Sendable {
    let dayLabel: String
    let sessionName: String
    let exercises: [PlannedExercise]
    var id: String { dayLabel }

    enum CodingKeys: String, CodingKey {
        case dayLabel = "day_label"
        case sessionName = "session_name"
        case exercises
    }

    let reps: String
    let restSeconds: Int
    let rationale: String

enum PlanPromptBuilder {
    static func buildSystemPrompt(profile: UserProfile) -> String {
        var prompt = """
        You are a professional fitness coach. Generate a personalized \(profile.daysPerWeek)-day weekly workout plan.

        User profile:
        - Goal: \(profile.goal)
        - Fitness Level: \(profile.fitnessLevel)
        - Training Days: \(profile.daysPerWeek) days per week
        - Available Equipment: \(profile.equipment.joined(separator: ", "))
        """

        if !profile.injuries.isEmpty {
            prompt += "\n- Areas to avoid or modify around: \(profile.injuries)"
        }

        prompt += """

        For each exercise, provide a rationale explaining why it was chosen for this user's specific goal and available equipment.

        SAFETY: You are not a medical professional. Do not diagnose conditions or prescribe treatments. If the user mentions pain or injury, recommend consulting a physician.
        """

        return prompt
    }
}

**WorkoutPlanRepository.swift** -- Save and fetch with rawJSON fallback:

```swift
import CoreData
import Foundation

@MainActor
final class WorkoutPlanRepository {
    private let context: NSManagedObjectContext

    init(context: NSManagedObjectContext = PersistenceController.shared.container.viewContext) {
        self.context = context
    }

    func save(plan: WorkoutPlan, supabaseId: String, userId: String) throws {
        let cdPlan = CDWorkoutPlan(context: context)
        cdPlan.id = UUID()
        cdPlan.supabaseId = supabaseId
        cdPlan.userId = userId
        cdPlan.planName = plan.planName
        cdPlan.goalSummary = plan.goalSummary
        cdPlan.rawJSON = try JSONEncoder().encode(plan)
        cdPlan.createdAt = Date()
        cdPlan.isActive = true

        for (index, day) in plan.weeklyDays.enumerated() {
            let cdDay = CDWorkoutDay(context: context)
            cdDay.id = UUID()
            cdDay.dayLabel = day.dayLabel
            cdDay.sessionName = day.sessionName
            cdDay.sortOrder = Int16(index)

            for (exIndex, exercise) in day.exercises.enumerated() {
                let cdExercise = CDPlannedExercise(context: context)
                cdExercise.id = UUID()
                cdExercise.exerciseName = exercise.exerciseName
                cdExercise.sets = Int16(exercise.sets)
                cdExercise.reps = exercise.reps
                cdExercise.restSeconds = Int32(exercise.restSeconds)
                cdExercise.rationale = exercise.rationale
                cdExercise.sortOrder = Int16(exIndex)
                cdDay.addToExercises(cdExercise)
            }
            cdPlan.addToDays(cdDay)
        }

        try context.save()
    }

    func fetchActivePlan(userId: String) throws -> WorkoutPlan? {
        let request = CDWorkoutPlan.fetchRequest()
        request.predicate = NSPredicate(format: "userId == %@ AND isActive == YES", userId)
        request.sortDescriptors = [NSSortDescriptor(key: "createdAt", ascending: false)]
        request.fetchLimit = 1

        guard let cdPlan = try context.fetch(request).first,
              let rawJSON = cdPlan.rawJSON else { return nil }

        // Prefer decoding from rawJSON (defensive fallback)
        return try JSONDecoder().decode(WorkoutPlan.self, from: rawJSON)
    }

    func deactivateAllPlans(userId: String) throws {
        let request = CDWorkoutPlan.fetchRequest()
        request.predicate = NSPredicate(format: "userId == %@ AND isActive == YES", userId)
        let plans = try context.fetch(request)
        for plan in plans {
            plan.isActive = false
        }
        if !plans.isEmpty {
            try context.save()
        }
    }
}
```
