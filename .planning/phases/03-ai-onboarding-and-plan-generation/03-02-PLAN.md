**OnboardingView.swift** — Root container presented as `.fullScreenCover`:
```swift
struct OnboardingView: View {
    @State private var viewModel = OnboardingViewModel()
    // Receives an onComplete callback from parent for plan generation trigger
    var onComplete: ((UserProfile) -> Void)?

    var body: some View {
        ZStack {
            Color("AppBackground").ignoresSafeArea()

            VStack(spacing: 0) {
                // Top: progress bar + back chevron
                HStack {
                    if viewModel.currentStep > 0 {
                        Button(action: { viewModel.goBack() }) {
                            Image(systemName: "chevron.left")
                                .font(.body)
                                .foregroundStyle(.primary)
                                .frame(width: 44, height: 44)
                        }
                        .accessibilityLabel("Go back")
                    } else {
                        // Placeholder for layout symmetry
                        Color.clear.frame(width: 44, height: 44)
                    }

                    Spacer()

                    OnboardingProgressView(
                        currentStep: viewModel.currentStep,
                        totalSteps: viewModel.totalSteps,
                        progress: viewModel.progressFraction
                    )

                    Spacer()
                    Color.clear.frame(width: 44, height: 44)  // balance
                }
                .padding(.horizontal, 16)  // md
                .padding(.top, 16)  // md below safe area

                // Card area
                ZStack {
                    // Use switch on currentStep with .id() and .transition()
                    // Each card gets the asymmetric slide transition
                }
                .frame(maxWidth: .infinity, maxHeight: .infinity)
            }
        }
        .confirmationDialog(
            "Quit setup?",
            isPresented: $viewModel.showQuitConfirmation,
            titleVisibility: .visible
        ) {
            Button("Quit", role: .destructive) { /* dismiss fullScreenCover */ }
            Button("Continue Setup", role: .cancel) { }
        } message: {
            Text("You'll lose your progress and won't get your personalized plan.")
        }
        .interactiveDismissDisabled()  // prevent accidental dismiss; force quit confirmation
        .onAppear {
            viewModel.onComplete = onComplete
        }
    }
}
```

Card transition: use `.transition()` with asymmetric move based on `viewModel.isGoingForward`:
- Forward: insertion `.move(edge: .trailing)`, removal `.move(edge: .leading)`
- Backward: insertion `.move(edge: .leading)`, removal `.move(edge: .trailing)`

Reduce Motion: check `@Environment(\.accessibilityReduceMotion) var reduceMotion`. If true, use `.opacity` transition instead of `.move`.

The ZStack card switching pattern: render only the current card based on `viewModel.currentStep` as a switch statement inside the ZStack. Apply `.id(viewModel.currentStep)` and `.transition(cardTransition)` to enable animation when step changes.
