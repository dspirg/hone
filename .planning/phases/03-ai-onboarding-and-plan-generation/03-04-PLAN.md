**PlanPreviewViewModel.swift** — Thin ViewModel bridging PlanGenerationService to UI:

```swift
@Observable
@MainActor
final class PlanPreviewViewModel {
    let service: PlanGenerationService
    private(set) var userProfile: UserProfile

    init(service: PlanGenerationService, profile: UserProfile) {
        self.service = service
        self.userProfile = profile
    }

    // Convenience computed properties for UI binding
    var isLoading: Bool {
        if case .streaming = service.state { return true }
        return service.state == .idle
    }

    var plan: WorkoutPlan? {
        if case .completed(let plan) = service.state { return plan }
        return nil
    }

    var streamingText: String {
        if case .streaming(let text) = service.state { return text }
        return ""
    }

    var errorMessage: String? {
        if case .error(let msg) = service.state { return msg }
        return nil
    }

    var regenerationsRemaining: Int { service.regenerationsRemaining }
    var canRegenerate: Bool { service.canRegenerate }

    var isStreaming: Bool {
        if case .streaming = service.state { return true }
        return false
    }

    func startGeneration() {
        Task {
            try? await service.saveProfile(userProfile)
            service.generatePlan(profile: userProfile)
        }
    }

    func regenerate() {
        service.regeneratePlan(profile: userProfile)
    }

    func retry() {
        service.generatePlan(profile: userProfile)
    }
}
```

```swift
struct WorkoutDayCardView: View {
    let day: WorkoutDay

    var body: some View {
        VStack(alignment: .leading, spacing: 0) {
            // Session name header — .title2 semibold
            Text(day.sessionName)
                .font(.title2.weight(.semibold))
                .padding(.horizontal, 16)  // md
                .padding(.top, 16)         // md
                .accessibilityAddTraits(.isHeader)

            // sm gap + divider
            Spacer().frame(height: 8)
            Divider().padding(.leading, 16)

            // Exercise rows separated by inset dividers
            ForEach(day.exercises) { exercise in
                ExerciseRowView(exercise: exercise)
                if exercise.id != day.exercises.last?.id {
                    Divider().padding(.leading, 16)
                }
            }

            Spacer().frame(height: 8)  // bottom padding
        }
        .background(Color("CardBackground"))
        .clipShape(RoundedRectangle(cornerRadius: 16))
        .padding(.horizontal, 16)  // md
    }
}
```
