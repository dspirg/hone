```swift
import SwiftUI

struct SessionSummaryView: View {
    let workoutDayLabel: String    // e.g., "Monday"
    let totalExercises: Int
    let totalSets: Int
    let totalReps: Int
    let duration: TimeInterval     // sessionDuration from SessionViewModel
    let onDone: () -> Void         // Dismisses SessionView

    var body: some View {
        ScrollView {
            VStack(spacing: 32) {
                Spacer().frame(height: 48)  // 2xl breathing room

                // Completion icon
                Image(systemName: "checkmark.circle.fill")
                    .font(.system(size: 72))
                    .foregroundStyle(Color("AccentColor"))
                    .accessibilityLabel("Session complete")

                // Headings
                VStack(spacing: 8) {
                    Text("Great work.")
                        .font(.title.weight(.semibold))

                    Text("\(workoutDayLabel) complete")
                        .font(.subheadline)
                        .foregroundStyle(.secondary)
                }

                // Stats row
                HStack(spacing: 32) {
                    StatCell(label: "Exercises", value: "\(totalExercises)")
                    StatCell(label: "Sets", value: "\(totalSets)")
                    StatCell(label: "Reps", value: "\(totalReps)")
                }
                .accessibilityElement(children: .combine)

                // Duration
                StatCell(label: "Duration", value: formattedDuration)

                Spacer()

                // Done button — no difficulty rating (Phase 8 scope per CONTEXT.md deferred)
                Button("Done", action: onDone)
                    .buttonStyle(.borderedProminent)
                    .frame(maxWidth: .infinity)
                    .frame(height: 52)
                    .padding(.horizontal, 16)
                    .padding(.bottom, 32)
            }
            .padding(.horizontal, 16)
        }
        .navigationBarBackButtonHidden(true)
        .background(Color("AppBackground").ignoresSafeArea())
    }

    // Duration format: "42m 07s"; under 1 minute: "0m 45s"
    private var formattedDuration: String {
        let total = Int(max(duration, 0))
        let minutes = total / 60
        let seconds = total % 60
        return "\(minutes)m \(String(format: "%02d", seconds))s"
    }
}

// MARK: - StatCell

struct StatCell: View {
    let label: String
    let value: String

    var body: some View {
        VStack(spacing: 4) {
            Text(value)
                .font(.title2.weight(.semibold))
            Text(label)
                .font(.subheadline)
                .foregroundStyle(.secondary)
        }
        .accessibilityElement(children: .combine)
    }
}
```

```swift
import SwiftUI
import CoreData

struct TrainView: View {
    @Environment(AppState.self) var appState
    @Environment(\.managedObjectContext) var context
    @State private var activePlan: WorkoutPlan?
    @State private var activePlanSupabaseId: String = ""
    @State private var isLoading = true

    var body: some View {
        NavigationStack {
            ScrollView {
                VStack(spacing: 16) {
                    if isLoading {
                        ProgressView().padding(.top, 48)
                    } else if let plan = activePlan {
                        // Active plan day cards
                        ForEach(plan.weeklyDays) { day in
                            workoutDayCard(day: day, planId: activePlanSupabaseId)
                        }

                        Divider().padding(.horizontal, 16)

                        // Browse exercises (Phase 2 preserved)
                        NavigationLink(destination: ExerciseLibraryView()) {
                            HStack {
                                Label("Browse Exercises", systemImage: "list.bullet")
                                    .font(.subheadline)
                                Spacer()
                                Image(systemName: "chevron.right")
                                    .font(.subheadline)
                                    .foregroundStyle(.secondary)
                            }
                            .padding(16)
                            .background(Color("CardBackground"))
                            .clipShape(RoundedRectangle(cornerRadius: 12))
                            .padding(.horizontal, 16)
                        }
                        .buttonStyle(.plain)

                    } else {
                        // No active plan (unexpected post-Phase-3 state)
                        VStack(spacing: 8) {
                            Image(systemName: "figure.run")
                                .font(.system(size: 48))
                                .foregroundStyle(.secondary)
                            Text("No workout planned for today.")
                                .font(.body)
                            Text("Your AI plan appears here.")
                                .font(.subheadline)
                                .foregroundStyle(.secondary)
                        }
                        .padding(.top, 48)

                        // Still offer exercise browsing
                        NavigationLink(destination: ExerciseLibraryView()) {
                            Label("Browse Exercises", systemImage: "list.bullet")
                                .font(.subheadline)
                                .padding(16)
                                .background(Color("CardBackground"))
                                .clipShape(RoundedRectangle(cornerRadius: 12))
                                .padding(.horizontal, 16)
                        }
                        .buttonStyle(.plain)
                        .padding(.top, 24)
                    }
                }
                .padding(.top, 16)
            }
            .navigationTitle("Train")
            .task { await loadActivePlan() }
        }
    }

    @ViewBuilder
    private func workoutDayCard(day: WorkoutDay, planId: String) -> some View {
        VStack(alignment: .leading, spacing: 12) {
            VStack(alignment: .leading, spacing: 4) {
                Text(day.dayLabel)
                    .font(.title2.weight(.semibold))
                Text(day.sessionName)
                    .font(.subheadline)
                    .foregroundStyle(.secondary)
                Text("\(day.exercises.count) exercises")
                    .font(.subheadline)
                    .foregroundStyle(.secondary)
            }

            NavigationLink {
                SessionView(workoutDay: day, planId: planId)
            } label: {
                Text("Start Workout")
                    .font(.subheadline.weight(.semibold))
                    .frame(maxWidth: .infinity)
                    .frame(height: 44)
            }
            .buttonStyle(.borderedProminent)
        }
        .padding(16)
        .background(Color("CardBackground"))
        .clipShape(RoundedRectangle(cornerRadius: 16))
        .padding(.horizontal, 16)
    }

    private func loadActivePlan() async {
        guard let userId = appState.currentUser?.id.uuidString else {
            isLoading = false
            return
        }
        do {
            let repo = WorkoutPlanRepository(context: context)
            activePlan = try repo.fetchActivePlan(userId: userId)

            // Fetch supabaseId from CDWorkoutPlan for use as planId in SessionView
            let cdReq = CDWorkoutPlan.fetchRequest()
            cdReq.predicate = NSPredicate(format: "userId == %@ AND isActive == YES", userId)
            cdReq.fetchLimit = 1
            if let cdPlan = try? context.fetch(cdReq).first {
                activePlanSupabaseId = cdPlan.supabaseId ?? ""
            }
        } catch {
            // Silent fail — show empty state
        }
        isLoading = false
    }
}
```
