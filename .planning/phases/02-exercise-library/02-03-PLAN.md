2. Create `VideoPlayerView.swift` — UIViewControllerRepresentable wrapping AVPlayerViewController:
   ```swift
   import SwiftUI
   import AVKit
   import MuxPlayerSwift

   struct VideoPlayerView: UIViewControllerRepresentable {
       let muxPlaybackId: String
       let localAssetURL: URL?

       func makeUIViewController(context: Context) -> AVPlayerViewController {
           let vc: AVPlayerViewController
           if let localURL = localAssetURL {
               // Offline path
               let player = AVPlayer(url: localURL)
               vc = AVPlayerViewController()
               vc.player = player
           } else {
               // Online path with Mux Smart Cache
               let options = PlaybackOptions(
                   enableSmartCache: true,
                   singleRenditionResolutionTier: .only720p
               )
               vc = AVPlayerViewController(
                   playbackID: muxPlaybackId,
                   playbackOptions: options
               )
           }
           // Setup looping via seek-to-zero (NEVER AVPlayerLooper for HLS)
           vc.player?.actionAtItemEnd = .none
           context.coordinator.setupLooping(player: vc.player)
           vc.player?.play()  // Auto-play on appear
           return vc
       }

       func updateUIViewController(_ vc: AVPlayerViewController, context: Context) {}

       func makeCoordinator() -> Coordinator { Coordinator() }

       class Coordinator: NSObject {
           private var observer: NSObjectProtocol?

           func setupLooping(player: AVPlayer?) {
               guard let player, let item = player.currentItem else { return }
               player.actionAtItemEnd = .none
               observer = NotificationCenter.default.addObserver(
                   forName: .AVPlayerItemDidPlayToEndTime,
                   object: item,
                   queue: .main
               ) { [weak player] _ in
                   player?.seek(to: .zero) { _ in
                       player?.play()
                   }
               }
           }

           deinit {
               if let observer { NotificationCenter.default.removeObserver(observer) }
           }
       }
   }
   ```

   CRITICAL ANTI-PATTERN: Do NOT use `AVPlayerLooper` anywhere. It is broken for HLS streams (causes duplicate downloads, erratic looping). The seek-to-zero notification pattern is the ONLY correct approach for Mux HLS.

1. Create `ExerciseDetailView.swift` per UI-SPEC:
   ```swift
   struct ExerciseDetailView: View {
       let exercise: ExerciseModel

       var body: some View {
           ScrollView {
               VStack(spacing: 0) {
                   // Video or placeholder (40% screen, 16:9 aspect)
                   if exercise.hasVideo, let playbackId = exercise.muxPlaybackId {
                       VideoPlayerView(
                           muxPlaybackId: playbackId,
                           localAssetURL: exercise.localAssetURL.flatMap { URL(string: $0) }
                       )
                       .aspectRatio(16/9, contentMode: .fit)
                   } else {
                       ExercisePlaceholderView(exerciseName: exercise.name)
                   }

                   // Metadata section
                   VStack(alignment: .leading, spacing: 16) {
                       // Exercise name
                       Text(exercise.name)
                           .font(.title2)
                           .fontWeight(.semibold)

                       // Muscle group tag pills
                       HStack(spacing: 8) {
                           TagPill(text: exercise.primaryMuscle, isAccent: true)
                           TagPill(text: exercise.equipmentTag, isAccent: false)
                           TagPill(text: exercise.difficulty, isAccent: false)
                       }

                       Divider()

                       // How-To Steps (numbered)
                       if !exercise.howToSteps.isEmpty {
                           Text("How To")
                               .font(.subheadline)
                               .fontWeight(.semibold)
                           VStack(alignment: .leading, spacing: 8) {
                               ForEach(Array(exercise.howToSteps.enumerated()), id: \.offset) { index, step in
                                   HStack(alignment: .top, spacing: 8) {
                                       Text("\(index + 1).")
                                           .font(.body)
                                           .foregroundStyle(.secondary)
                                       Text(step)
                                           .font(.body)
                                   }
                               }
                           }
                       }

                       // Form Tips
                       if let tips = exercise.formTips, !tips.isEmpty {
                           Divider()
                           Text("Form Tips")
                               .font(.subheadline)
                               .fontWeight(.semibold)
                           Text(tips)
                               .font(.subheadline)
                               .foregroundStyle(.secondary)
                       }
                   }
                   .padding(.horizontal, 32)  // xl spacing per UI-SPEC
                   .padding(.top, 24)          // lg spacing below video
               }
           }
           .navigationTitle(exercise.name)
           .navigationBarTitleDisplayMode(.inline)
       }
   }

   // Helper: tag pill component
   private struct TagPill: View {
       let text: String
       let isAccent: Bool

       var body: some View {
           Text(text)
               .font(.subheadline)
               .padding(.horizontal, 10)
               .padding(.vertical, 4)
               .background(isAccent ? Color("AccentColor") : Color("CardBackground"))
               .foregroundStyle(isAccent ? .white : .primary)
               .clipShape(Capsule())
       }
   }
   ```
