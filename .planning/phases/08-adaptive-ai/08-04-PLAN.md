2. Create `WorkoutApp/Features/Adaptation/MissedSessionDetector.swift`:
```swift
import Foundation

/// Detects planned-but-skipped training days for the current week (D-06, D-07).
/// Compares active plan day labels against completed CDSessionLog records.
/// Only flags planned training days — NOT rest days.
struct MissedSessionDetector {
    static func detectMissedSessions(
        activePlanDayLabels: [String],
        completedSessions: [CDSessionLog],
        today: Date = Date(),
        calendar: Calendar = .current
    ) -> [String] {
        guard let weekInterval = calendar.dateInterval(of: .weekOfYear, for: today)
        else { return [] }

        // Day labels of completed sessions this week
        let completedLabels = Set(
            completedSessions
                .filter { s in
                    guard let completedAt = s.completedAt else { return false }
                    return weekInterval.contains(completedAt)
                }
                .compactMap { $0.workoutDayLabel }
        )

        // Map day labels to day-of-week index for "is this day in the past?" check
        let dayOfWeekMap: [String: Int] = [
            "Monday": 2, "Tuesday": 3, "Wednesday": 4,
            "Thursday": 5, "Friday": 6, "Saturday": 7, "Sunday": 1
        ]

        let todayWeekday = calendar.component(.weekday, from: today)

        return activePlanDayLabels.filter { label in
            // Only flag if: (a) not completed, (b) the planned day is in the past
            guard !completedLabels.contains(label) else { return false }
            guard let plannedWeekday = dayOfWeekMap[label] else { return false }
            return plannedWeekday < todayWeekday
        }
    }
}
```

```swift
import Foundation

/// Calls adapt-plan and regenerate-plan Edge Functions.
/// Injected via @Environment at MainTabView level (same pattern as SessionSyncService).
@Observable @MainActor
final class AdaptationService {
    private let supabaseURL: String
    private let supabaseAnonKey: String
    private var lastWeeklyCheck: String = "" // ISO week key to prevent duplicate Monday checks

    init() {
        self.supabaseURL = // read from Config/Supabase.plist or environment
        self.supabaseAnonKey = // read from Config/Supabase.plist or environment
    }

    /// Called after session completion + rating (ADPT-01).
    func requestPostSessionAdaptation(
        rating: DifficultyRating,
        accessToken: String
    ) async throws -> AdaptedPlanResponse {
        let body = AdaptPlanRequest(
            triggerType: "post_session",
            currentRating: rating.rawValue,
            missedSessions: nil
        )
        return try await callEdgeFunction(
            path: "adapt-plan",
            body: body,
            accessToken: accessToken
        )
    }

    /// Called when missed sessions detected on app foreground (ADPT-03).
    func requestMissedSessionAdaptation(
        missedDays: [String],
        accessToken: String
    ) async throws -> AdaptedPlanResponse {
        let body = AdaptPlanRequest(
            triggerType: "missed_session",
            currentRating: nil,
            missedSessions: missedDays
        )
        return try await callEdgeFunction(
            path: "adapt-plan",
            body: body,
            accessToken: accessToken
        )
    }

    /// Called on Monday morning app foreground (ADPT-02, D-04).
    func requestWeeklyRegeneration(
        accessToken: String
    ) async throws -> AdaptedPlanResponse {
        return try await callEdgeFunction(
            path: "regenerate-plan",
            body: AdaptPlanRequest(triggerType: "weekly", currentRating: nil, missedSessions: nil),
            accessToken: accessToken
        )
    }

    /// Foreground check: weekly regen on Monday + missed session detection (D-04, D-07).
    func checkOnForeground(
        accessToken: String,
        activePlanDayLabels: [String],
        completedSessions: [CDSessionLog]
    ) async {
        // Weekly regeneration check — Monday only, once per ISO week
        let calendar = Calendar.current
        let weekday = calendar.component(.weekday, from: Date())
        let isoWeek = ISO8601DateFormatter().string(from: Date()).prefix(8) // "2026-W17" approx
        if weekday == 2 && String(isoWeek) != lastWeeklyCheck { // Monday = weekday 2
            lastWeeklyCheck = String(isoWeek)
            do {
                let _ = try await requestWeeklyRegeneration(accessToken: accessToken)
            } catch {
                print("AdaptationService: weekly regeneration failed: \(error)")
            }
        }

        // Missed session detection
        let missed = MissedSessionDetector.detectMissedSessions(
            activePlanDayLabels: activePlanDayLabels,
            completedSessions: completedSessions
        )
        if !missed.isEmpty {
            do {
                let _ = try await requestMissedSessionAdaptation(
                    missedDays: missed,
                    accessToken: accessToken
                )
            } catch {
                print("AdaptationService: missed session adaptation failed: \(error)")
            }
        }
    }

    private func callEdgeFunction<T: Encodable>(
        path: String,
        body: T,
        accessToken: String
    ) async throws -> AdaptedPlanResponse {
        guard let url = URL(string: "\(supabaseURL)/functions/v1/\(path)") else {
            throw URLError(.badURL)
        }
        var request = URLRequest(url: url)
        request.httpMethod = "POST"
        // Manual auth headers — same Supabase SDK bug #634 as PlanSSEClient
        request.setValue("Bearer \(accessToken)", forHTTPHeaderField: "Authorization")
        request.setValue(supabaseAnonKey, forHTTPHeaderField: "apikey")
        request.setValue("application/json", forHTTPHeaderField: "Content-Type")
        request.httpBody = try JSONEncoder().encode(body)

        let (data, response) = try await URLSession.shared.data(for: request)
        guard let httpResponse = response as? HTTPURLResponse,
              httpResponse.statusCode == 200 else {
            throw URLError(.badServerResponse)
        }
        return try JSONDecoder().decode(AdaptedPlanResponse.self, from: data)
    }
}
```
